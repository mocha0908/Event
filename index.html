// 1. データ取得と画面描画（時差バケ修正・日付フォーマット変更版） async function fetchEvents() { const loadingEl = document.getElementById('loading'); const listEl = document.getElementById('event-list'); loadingEl.style.display = 'block'; listEl.innerHTML = ''; try { const response = await fetch(GAS_API_URL); const data = await response.json(); // 💡 日本時間（JST）ベースで「今日」と「明日」の文字列（YYYY-MM-DD）を作る const getJstDateString = (dateObj) => { const y = dateObj.getFullYear(); const m = String(dateObj.getMonth() + 1).padStart(2, '0'); const d = String(dateObj.getDate()).padStart(2, '0'); return `${y}-${m}-${d}`; }; const now = new Date(); const todayStr = getJstDateString(now); // 例: "2026-06-11" const tomorrow = new Date(); tomorrow.setDate(now.getDate() + 1); const tomorrowStr = getJstDateString(tomorrow); // 例: "2026-06-12" // 💡 日付を「XXXX年XX月XX日」に変換するヘルパー関数 const formatToJapaneseDate = (dateStr) => { if (!dateStr) return '未設定'; const parts = dateStr.split('-'); if (parts.length !== 3) return dateStr; return `${parts[0]}年${Number(parts[1])}月${Number(parts[2])}日`; }; // 💡 締切優先の2段階ソートロジック const sortedData = data.sort((a, b) => { const aIsUrgent = (a.申し込み締め切り === todayStr || a.申し込み締め切り === tomorrowStr); const bIsUrgent = (b.申し込み締め切り === todayStr || b.申し込み締め切り === tomorrowStr); // どちらも今日・明日締め切りの場合は、締切日が近い方を上に if (aIsUrgent && bIsUrgent) { return new Date(a.申し込み締め切り) - new Date(b.申し込み締め切り); } // 片方だけが今日・明日締め切りなら、それを最優先で上に if (aIsUrgent && !bIsUrgent) return -1; if (!aIsUrgent && bIsUrgent) return 1; // 通常データ同士は「開催日」が近い順 return new Date(a.開催日) - new Date(b.開催日); }); loadingEl.style.display = 'none'; sortedData.forEach(event => { const isUrgent = (event.申し込み締め切り === todayStr || event.申し込み締め切り === tomorrowStr); const borderClass = isUrgent ? "border-2 border-red-500 bg-red-50/30 animate-pulse-subtle" : "border-gray-200 bg-white"; // 番号データのパース const p1 = parseNumberCell(event.番号1); const p2 = parseNumberCell(event.番号2); const p3 = parseNumberCell(event.番号3); const displayNumbers = [p1, p2, p3] .filter(p => p.num !== "") .map(p => `${p.num}(${p.status.replace('使用','').replace('み','')})`) .join(", "); // 日付の表示形式を「年月日」に変換 const displayHoldingDate = formatToJapaneseDate(event.開催日); const displayDeadlineDate = formatToJapaneseDate(event.申し込み締め切り); const card = document.createElement('div'); card.className = `${borderClass} p-5 rounded-xl shadow-md flex flex-col justify-between space-y-4`; card.innerHTML = `
${displayHoldingDate} ${event.申し込み締め切り ? ` ${event.申し込み締め切り === todayStr ? '⚠️ 今日締切！' : event.申し込み締め切り === tomorrowStr ? '⏳ 明日締切！' : '締切: ' + displayDeadlineDate} ` : ''}

${isUrgent ? '🔥 ' : ''}${event.イベント名 || '未設定'}

📍 ${event.会場 || '未設定'} (枠: ${event.枠数 || '-'})

${event.URL ? `
🔗 公式URL: ${event.URL}
` : ''}
管理番号: ${displayNumbers || '未入力'}
番:
番:
`; listEl.appendChild(card); }); } catch (error) { loadingEl.textContent = "データの読み込みに失敗しました。"; } }

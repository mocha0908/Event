// 1. データ取得と画面描画（締切間近を最優先ソート） async function fetchEvents() { const loadingEl = document.getElementById('loading'); const listEl = document.getElementById('event-list'); loadingEl.style.display = 'block'; listEl.innerHTML = ''; try { const response = await fetch(GAS_API_URL); const data = await response.json(); // 💡 今日の日付を特定するための準備（日付比較用） const now = new Date(); const todayStr = now.toISOString().split('T')[0]; // "2026-06-11" のような形式 const tomorrow = new Date(); tomorrow.setDate(now.getDate() + 1); const tomorrowStr = tomorrow.toISOString().split('T')[0]; // 💡 特殊ソートロジック const sortedData = data.sort((a, b) => { // 申し込み締め切りが今日・明日かどうかを判定 const aIsUrgent = (a.申し込み締め切り === todayStr || a.申し込み締め切り === tomorrowStr); const bIsUrgent = (b.申し込み締め切り === todayStr || b.申し込み締め切り === tomorrowStr); // どちらも緊急、またはどちらも緊急ではない場合は「開催日」が近い順 if (aIsUrgent && bIsUrgent) { return new Date(a.申し込み締め切り) - new Date(b.申し込み締め切り); // 締切が今日の方をより上に } // aだけが今日・明日締め切りなら、aを上にする if (aIsUrgent && !bIsUrgent) return -1; // bだけが今日・明日締め切りなら、bを上にする if (!aIsUrgent && bIsUrgent) return 1; // 通常データ同士は「開催日」が近い順（これまでのロジック） return new Date(a.開催日) - new Date(b.開催日); }); loadingEl.style.display = 'none'; sortedData.forEach(event => { // 💡 締め切りが今日か明日ならカードに赤い枠線をつけてさらに目立たせる const isUrgent = (event.申し込み締め切り === todayStr || event.申し込み締め切り === tomorrowStr); const borderClass = isUrgent ? "border-2 border-red-500 bg-red-50/30 animate-pulse-subtle" : "border-gray-200 bg-white"; const numbers = [event.番号1, event.番号2, event.番号3, event.番号4, event.番号5] .filter(num => num !== undefined && num !== null && num !== "") .join(", "); const card = document.createElement('div'); // borderClass を適用して見た目も差別化 card.className = `${borderClass} p-6 rounded-xl shadow-md flex flex-col justify-between space-y-4`; card.innerHTML = `
${event.開催日} ${event.申し込み締め切り ? ` ${event.申し込み締め切り === todayStr ? '⚠️ 今日締切！' : event.申し込み締め切り === tomorrowStr ? '⏳ 明日締切！' : '締切: ' + event.申し込み締め切り} ` : ''}

${isUrgent ? '🔥 ' : ''}${event.イベント名 || '未設定'}

📍 ${event.会場 || '未設定'} (枠数: ${event.枠数 || '-'})

${event.URL ? `🔗 公式リンクへ飛ぶ` : ''}
管理番号: ${numbers || '未入力'}
番:
番:
`; listEl.appendChild(card); }); } catch (error) { loadingEl.textContent = "データの読み込みに失敗しました。"; } }
